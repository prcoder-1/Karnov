struct MOVFile {                                                                                                                                                                                                                         
      uint16_t  start_record_idx;  /* LE; в файлах всегда 0 — это "текущий индекс при старте" */                                                                                                                                             
      uint16_t  num_records;       /* LE; число записанных изменений ввода */                                                                                                                                                                
      struct {                                                                                                                                                                                                                               
          uint8_t   input_state;   /* битовая маска */                                                                                                                                                                                       
          uint16_t  frame_number;  /* LE; номер игрового кадра, на котором ввод становится активен */

  };

Битовая маска input_state:                                                                                                                                                                   
  
  ┌───────────┬───────────────────────────────┬──────────────────┐                                                                                                                                                                           
  │    Бит    │           Значение            │ Привязка клавиш  │                                                                                                                                                                         
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x01      │ RIGHT                         │ Num6             │
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x02      │ LEFT                          │ Num4             │                                                                                                                                                                         
  ├───────────┼───────────────────────────────┼──────────────────┤
  │ 0x04      │ DOWN                          │ Num2             │
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x08      │ UP                            │ Num8             │
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x10      │ FIRE                          │ X / Space / Btn1 │                                                                                                                                                                         
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x20      │ OPTION (использовать предмет) │ Z / Btn2         │
  ├───────────┼───────────────────────────────┼──────────────────┤                                                                                                                                                                           
  │ 0x40/0x80 │ не используются               │ —                │                                                                                                                                                                         
  └───────────┴───────────────────────────────┴──────────────────┘                                                                                                                                                                           
                                                                                                                                                                                                                                           
  Диагонали — комбинации (Num1=0x06=LEFT+DOWN, Num3=0x05=RIGHT+DOWN, Num7=0x0A=LEFT+UP, Num9=0x09=RIGHT+UP). 

Логика воспроизведения (по cs:0x6dc0): для каждого кадра игры читается текущая запись; если её frame_number ≤ current_frame, то её input_state копируется в глобальное состояние
ввода, а индекс инкрементируется. Демо считается  завершённым, когда curr_idx == num_records — после этого, если уровень ≠ 9 (последний), возвращается признак «конец демо» и игра выходит из аттракт-режима.

Формат LEVEL%d.BIN (данные уровня)
  1. RLE-сжатие с маркером 0x5A (буква Z)                                                                                                                                                                                                    
                                                                                                                                                                                                                                             
  Алгоритм декодирования (из cs:0x7127):                                                                                                                                                                                                     
                                                                                                                                                                                                                                           
  def rle_decompress(data: bytes) -> bytes:                                                                                                                                                                                                  
      out = bytearray(); i = 0                                                                                                                                                                                                             
      while i < len(data):
          b = data[i]; i += 1
          if b != 0x5A:                                                                                                                                                                                                                      
              out.append(b); continue
          n = data[i]; i += 1                                                                                                                                                                                                                
          if n == 0x5A:                  # 5A 5A  -> литерал 0x5A                                                                                                                                                                          
              out.append(0x5A); continue                                                                                                                                                                                                     
          v = data[i]; i += 1            # 5A N B -> байт B повторён (N или 256, если N==0) раз
          out.extend([v] * (256 if n == 0 else n))                                                                                                                                                                                           
      return bytes(out)                                                                                                                                                                                                                    
                                                                                                                                                                                                                                             
  Поэтому в файлах так часто видны байты 5A — это служебные escape-маркеры серий.

Структура распакованного потока                                                                                                                                                                                                         
                                                                                                                                                                                                                                           
  Загрузчик читает следующие секции последовательно (через декомпрессирующий fread):                                                                                                                                                         
  
  /* (A) Таблица "областей/экранов" уровня (call cs:0x51b6) */                                                                                                                                                                               
  uint16_t  num_objects;                                                                                                                                                                                                                     
  struct ObjectRegion {
      uint16_t  x0, y0;      /* левый-верх области */                                                                                                                                                                                        
      uint16_t  x1, y1;      /* правый-низ */                                                                                                                                                                                                
      uint16_t  vx, vy;      /* скорость скролла или связанная точка */                                                                                                                                                                      
  } objects[num_objects];    /* 12 байт; первое (x1,y1) копируется в [0x48c]/[0x48e] */                                                                                                                                                      
                                                                                                                                                                                                                                             
  /* (B) Таблица спавна врагов/объектов (call cs:0x5216) */                                                                                                                                                                                  
  uint16_t  num_enemies;                                                                                                                                                                                                                     
  struct EnemySpawn {                                                                                                                                                                                                                        
      uint16_t  type;        /* идентификатор спрайта */                                                                                                                                                                                   
      uint16_t  subtype;     /* вариант / поведение */                                                                                                                                                                                       
      uint16_t  cur_x, cur_y;                                                                                                                                                                                                                
      uint16_t  range_x, range_y;                                                                                                                                                                                                            
      uint16_t  spawn_x, spawn_y;                                                                                                                                                                                                            
      uint16_t  trigger;     /* условие появления */                                                                                                                                                                                         
      uint16_t  flags;                                                                                                                                                                                                                       
  } enemies[num_enemies];    /* 20 байт */
                                                                                                                                                                                                                                             
  /* (C) Заголовок карты */                                                                                                                                                                                                                
  uint16_t  num_map_rows;    /* верхняя граница цикла построчного чтения */
  uint16_t  num_unique_tiles;/* [0xb830] — число уникальных тайлов */                                                                                                                                                                        
  uint16_t  tile_size_words; /* [0xcc96] — размер одного тайла в WORD-ах */                                                                                                                                                                  
                                                                                                                                                                                                                                             
  /* (D) Описания строк карты (повторяется num_map_rows раз) */                                                                                                                                                                              
  struct MapRow {                                                                                                                                                                                                                            
      uint8_t   tile_idx;                 /* индекс в таблице тайлов */                                                                                                                                                                      
      uint8_t   attr;                     /* атрибут (палитра/поведение) */
      uint8_t   row_data[40 or 80];       /* столбцы; 80 байт для режима [0x360]==3 */                                                                                                                                                       
  };                                                                                                                                                                                                                                         
                                                                                                                                                                                                                                             
  /* (E) Графика тайлов */                                                                                                                                                                                                                   
  uint8_t   tile_pixels[num_unique_tiles * tile_size_words * 2];                                                                                                                                                                             
                                                                                                                                                                                                                                           
  После каждой строки атрибут записывается в lookup-таблицу DS:[tile_idx*2 - 0x5440], а сами 40/80 байт данных тайла попадают по дальнему указателю DS:[0x98] + tile_idx * size.
